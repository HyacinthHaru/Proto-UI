import {createRequire} from 'node:module';
import {writeFile} from 'node:fs/promises';
import assert from 'node:assert/strict';
const require=createRequire(process.env.PUI_WWW_PACKAGE);
const {chromium}=require('playwright-core');
const browser=await chromium.launch({executablePath:process.env.CHROME_PATH});
const rows=[];
const failures=[];
try {
  const page=await browser.newPage();
  if(process.env.PROTO_UI_BROWSER_BASE_URL){
    const url=process.env.PROTO_UI_BROWSER_BASE_URL;
    await page.route(url+'/__native-content-lease__',route=>route.fulfill({
      contentType:'text/html',body:'<!doctype html><main>Native content lease fixture</main>',
    }));
    await page.goto(url+'/__native-content-lease__');
  }else await page.setContent('<main></main>');
  await page.addScriptTag({path:process.argv[2]});
  if(process.env.PUI_PACKED_LEASE_ASSET){
    await page.evaluate(async url=>{
      const entry=await import(url);
      if(typeof entry.w!=='function'||typeof entry.r!=='function')throw new Error('Unexpected native lease export binding');
      window.nativeLease.candidate={withNativeContentLease:entry.w,registerNativeContentContainer:entry.r};
    },process.env.PROTO_UI_BROWSER_BASE_URL+'/_astro/'+process.env.PUI_PACKED_LEASE_ASSET);
  }
  const cdp=await page.context().newCDPSession(page);
  await cdp.send('Performance.enable');
  const metric=async()=>Object.fromEntries((await cdp.send('Performance.getMetrics')).metrics.map(m=>[m.name,m.value]));
  for(const mode of ['empty','forward','backward','crossing','registered-forward','registered-backward','input-forward','input-backward','textarea-forward','textarea-backward','input-move-update','input-focus-update','textarea-move-update','textarea-focus-update','shadow-input-backward','shadow-input-focus-update','shadow-textarea-backward','shadow-textarea-focus-update','shadow-input-other-focus','shadow-input-other-then-blur','shadow-textarea-other-focus','shadow-textarea-other-then-blur','input-connected-blur','textarea-connected-blur','input-same-parent','shadow-input-same-parent','host-descendant'])
    for(const variant of ['legacy','candidate']) {
      const before=await page.evaluate(({mode,variant})=>{
        const selection=document.getSelection();
        selection.removeAllRanges();
        const from=document.createElement('div'), destination=document.createElement('div');
        const source=document.createElement('span'), text=document.createTextNode('Retained authored source');
        const input=document.createElement(mode.includes('textarea')?'textarea':'input');input.value='Uncontrolled input';
        if(mode.startsWith('shadow-')||mode==='host-descendant'){const host=document.createElement('div');if(mode==='host-descendant')host.tabIndex=0;host.attachShadow({mode:'open'}).append(input);source.append(text,host);}else source.append(text,input);from.append(source);
        const outside=document.createTextNode('Other owner text');
        const other=document.createElement('input');other.value='Other focus owner';
        const marker=document.createElement('div'); marker.textContent='Layout witness';marker.style.width='100px';
        document.body.replaceChildren(from,outside,destination,marker,other);
        if(mode.includes('input-')||mode.includes('textarea-')){input.focus();input.setSelectionRange(3,8,mode.endsWith('backward')?'backward':'forward');}
        if(mode==='host-descendant')input.getRootNode().host.focus();
        if(mode==='forward'||mode==='backward') selection.setBaseAndExtent(text,mode==='forward'?2:10,text,mode==='forward'?10:2);
        if(mode==='crossing')selection.setBaseAndExtent(text,2,outside,5);
        if(mode.startsWith('registered')) {
          window.nativeLease[variant==='legacy'?'control':'candidate'].registerNativeContentContainer(from,[source]);
          selection.setBaseAndExtent(from,mode.endsWith('forward')?0:1,from,mode.endsWith('forward')?1:0);
        }
        if(mode==='no-api')destination.moveBefore=undefined;
        const update=()=>{input.value='New text';input.setSelectionRange(1,4,'backward');};
        if(mode.endsWith('focus-update'))input.addEventListener('focus',update);
        window.__atomicFixture={from,destination,source,text,outside,input,other,mode,update};
        const old=Document.prototype.getSelection;
        const native=Element.prototype.moveBefore;
        const count=window.__nativeLeaseCounts={selection:0,atomic:0,track:false};
        Document.prototype.getSelection=function(){if(count.track)count.selection++;return old.call(this);};
        if(native)Element.prototype.moveBefore=function(...args){if(count.track)count.atomic++;return Reflect.apply(native,this,args);};
        window.__restoreNative=()=>{Document.prototype.getSelection=old;if(native)Element.prototype.moveBefore=native;};
        // Resolve setup layout before the following deliberate dirty write.
        marker.getBoundingClientRect();
        marker.style.width='101px';
        return {mode,hasNative:typeof native==='function'};
      },{mode,variant});
      const statsBefore=await metric();
      const operation=await page.evaluate(({variant})=>{
        const {source,destination}=window.__atomicFixture,c=window.__nativeLeaseCounts;
        c.track=true;
        try {
          window.nativeLease[variant==='legacy'?'control':'candidate'].withNativeContentLease(source,()=>{if(window.__atomicFixture.mode.endsWith('connected-blur')){window.__atomicFixture.input.blur();return;}if(window.__atomicFixture.mode.endsWith('same-parent'))window.__atomicFixture.from.insertBefore(source,null);else destination.insertBefore(source,null);if(window.__atomicFixture.mode==='host-descendant'){window.__atomicFixture.input.focus();window.__atomicFixture.input.setSelectionRange(1,4,'backward');}if(window.__atomicFixture.mode.endsWith('move-update'))window.__atomicFixture.update();if(window.__atomicFixture.mode.endsWith('other-focus')){window.__atomicFixture.other.focus();window.__atomicFixture.other.setSelectionRange(1,4,'backward');}if(window.__atomicFixture.mode.endsWith('other-then-blur')){window.__atomicFixture.other.focus();window.__atomicFixture.other.blur();}});
        } finally {c.track=false;}
        return {selectionReads:c.selection,nativeMoves:c.atomic};
      },{variant});
      const statsAfter=await metric();
      const result=await page.evaluate(({mode})=>{
        const {source,text,outside,input,other,from,destination}=window.__atomicFixture;
        const sel=document.getSelection();
        const name=node=>node===text?'text':node===outside?'outside':node===destination?'destination':node?.nodeName??null;
        let active=document.activeElement;while(active?.shadowRoot?.activeElement)active=active.shadowRoot.activeElement;
        const out={sourceRetained:(mode.endsWith('same-parent')||mode.endsWith('connected-blur')?from:destination).firstChild===source,activeInput:active===input,activeOther:active===other,activeTag:active?.tagName??null,otherValue:other.value,otherSelection:[other.selectionStart,other.selectionEnd,other.selectionDirection],inputValue:input.value,
          selectionStart:input.selectionStart,selectionEnd:input.selectionEnd,selectionDirection:input.selectionDirection,
          anchor:name(sel.anchorNode),anchorOffset:sel.anchorOffset,focus:name(sel.focusNode),focusOffset:sel.focusOffset};
        window.__restoreNative();
        return out;
      },{mode});
      const row={mode,variant,before,operation,layoutDelta:statsAfter.LayoutCount-statsBefore.LayoutCount,result};
      rows.push(row);
      console.info(mode,variant,JSON.stringify({operation,layoutDelta:row.layoutDelta,result}));
      try {
      assert(result.sourceRetained);
      if(mode==='host-descendant'){assert(result.activeInput);assert.equal(result.inputValue,'Uncontrolled input');assert.deepEqual([result.selectionStart,result.selectionEnd,result.selectionDirection],[1,4,'backward']);}
      else if(mode.endsWith('connected-blur')){assert(!result.activeInput&&!result.activeOther);assert.equal(result.activeTag,'BODY');}
      else if(mode.endsWith('other-focus')){assert(!result.activeInput&&result.activeOther);assert.equal(result.otherValue,'Other focus owner');assert.deepEqual(result.otherSelection,[1,4,'backward']);}
      else if(mode.endsWith('other-then-blur')){assert(!result.activeInput&&!result.activeOther);assert.equal(result.activeTag,'BODY');}
      else if(mode.includes('input-')||mode.includes('textarea-')){
        const updated=mode.endsWith('-update');
        assert(result.activeInput&&result.inputValue===(updated?'New text':'Uncontrolled input'));
        assert.deepEqual([result.selectionStart,result.selectionEnd,result.selectionDirection],
          updated?[1,4,'backward']:[3,8,mode.endsWith('backward')?'backward':'forward']);
      }
      if(mode==='forward'||mode==='backward')
        assert.deepEqual([result.anchor,result.anchorOffset,result.focus,result.focusOffset],
          ['text',mode==='forward'?2:10,'text',mode==='forward'?10:2]);
      if(mode==='crossing') assert.deepEqual([result.anchor,result.anchorOffset,result.focus,result.focusOffset],['text',2,'outside',5]);
      if(mode.startsWith('registered'))
        assert.deepEqual([result.anchor,result.anchorOffset,result.focus,result.focusOffset],
          ['destination',mode.endsWith('forward')?0:1,'destination',mode.endsWith('forward')?1:0]);
      } catch(error) { failures.push({mode,variant,message:error.message}); }
    }
  await writeFile(process.argv[3],JSON.stringify({browser:browser.version(),rows,failures},null,2)+'\n');
  if(failures.some(row=>row.variant==='candidate')) {
    console.info('Retained failures', JSON.stringify(failures));
    process.exitCode=1;
  }
} finally {await browser.close();}
