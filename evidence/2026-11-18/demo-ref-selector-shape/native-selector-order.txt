import { createRequire, stripTypeScriptTypes } from 'node:module';
import { readFile, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import assert from 'node:assert/strict';

const [packagePath, parentPath, candidatePath, output] = process.argv.slice(2);
const { chromium } = createRequire(packagePath)('playwright-core');
const parent = await readFile(parentPath, 'utf8');
const candidate = await readFile(candidatePath, 'utf8');
const browser = await chromium.launch({ executablePath: process.env.CHROME_PATH });
const digest = (source) => createHash('sha256').update(source).digest('hex');
const results = [];
try {
  for (const implementation of ['parent', 'candidate', 'candidate', 'parent']) {
    const source = implementation === 'parent' ? parent : candidate;
    const page = await browser.newPage();
    try {
      await page.setContent('<main></main>');
      await page.addScriptTag({
        type: 'module',
        content: stripTypeScriptTypes(source) +
          '\nwindow.collectForProbe = collectOwnedDemoRefs;',
      });
      await page.waitForFunction(() => typeof window.collectForProbe === 'function');
      const result = await page.evaluate(() => {
        const main = document.querySelector('main');
        const collect = window.collectForProbe;
        function node(id, ref, owner, mode) {
          const element = document.createElement('div');
          element.id = id;
          if (ref !== null) element.dataset.demoRef = ref;
          if (mode !== 'class') element.dataset.demoOwner = owner;
          if (mode !== 'data') element.className = 'pui-owner-' + owner;
          return element;
        }
        const owner = 'demo-external';
        const before = node('before-host', 'item', owner, 'data');
        const host = node('host', 'root', owner, 'class');
        host.append(node('local-data', 'item', owner, 'data'), node('local-both', 'unique', owner, 'both'));
        const after = node('after-host', 'other', owner, 'class');
        main.append(before, host, after, node('missing-ref', null, owner, 'both'));
        const hidden = document.createElement('div');
        hidden.attachShadow({ mode: 'closed' }).append(node('hidden', 'other', owner, 'both'));
        main.append(hidden);
        const snapshot = (root) => {
          const found = collect(root, owner);
          return {
            ordered: [...found.nodes].map((entry) => entry.id),
            selected: Object.fromEntries(Object.entries(found.refs).map(([ref, entry]) => [ref, entry.id])),
          };
        };
        const attached = snapshot(host);
        host.remove();
        const detached = snapshot(host);
        before.removeAttribute('data-demo-owner');
        after.dataset.demoOwner = owner;
        after.className = 'pui-owner-someone-else';
        const mutated = snapshot(host);
        main.replaceChildren();
        const hosts = Array.from({length: 400}, (_, index) => {
          const owner = 'demo-' + index;
          const host = document.createElement('div');
          const noise = document.createDocumentFragment();
          for (let i = 0; i < 40; i++) noise.append(document.createElement('span'));
          host.append(noise, node('u-' + index, 'item', owner, index % 2 ? 'class' : 'data'));
          main.append(host);
          if (index % 15 === 0) main.prepend(node('p-' + index, 'portal', owner, 'data'));
          return { host, owner };
        });
        let checks = 0;
        let portal = 0;
        const start = performance.now();
        for (const {host, owner} of hosts) {
          const {refs} = collect(host, owner);
          if (refs.item) checks++;
          if (refs.portal) portal++;
        }
        return {attached, detached, mutated, checks, portal, durationMs: performance.now()-start};
      });
      results.push({implementation, ...result});
    } finally {
      await page.close();
    }
  }
  await writeFile(output, JSON.stringify({chrome:browser.version(), source:{parent:digest(parent),candidate:digest(candidate)},results},null,2)+'\n');
  for (const row of results) {
    assert.deepEqual(row.attached,{ordered:['local-data','local-both','before-host','host','after-host'],selected:{item:'before-host',unique:'local-both',root:'host',other:'after-host'}});
    assert.deepEqual(row.detached,{ordered:['local-data','local-both','before-host','after-host','host'],selected:{item:'before-host',unique:'local-both',root:'host',other:'after-host'}});
    assert.deepEqual(row.mutated,{ordered:['local-data','local-both','after-host','host'],selected:{item:'local-data',unique:'local-both',root:'host',other:'after-host'}});
    assert.equal(row.checks,400);
    assert.equal(row.portal,27);
  }
  console.log(results.map(({implementation,durationMs})=>({implementation,durationMs})));
} finally {
  await browser.close();
}
