from hashlib import sha256
from pathlib import Path
import json,sys
q,dist=map(Path,sys.argv[1:])
asset=dist/'site-shadcn-controls.DrSrHXZe.js'
before=asset.read_text()
source=before
def replace(old,new):
    global source
    assert source.count(old)==1,(old,source.count(old))
    source=source.replace(old,new,1)
prefix=r"""const puiRouterBindings={routers:0, listen:{}, nativeBridge:{}, protoDemand:{}, reset:{}};
function puiBindingCount(group,key,time){
 const item=puiRouterBindings[group][key]??={count:0,ms:0}; item.count++; item.ms+=time;
}
function puiRouterBindingCheckpoint(){
 if(++puiRouterBindings.routers%256===0)
  console.info("[S4 router bindings] "+JSON.stringify({...puiRouterBindings,pageMs:performance.now(),timeOrigin:performance.timeOrigin}));
}
"""
replace('function Qe(t,e,n){return t.addEventListener(e,n),()=>t.removeEventListener(e,n)}',
'function Qe(t,e,n){const started=performance.now();t.addEventListener(e,n);puiBindingCount("listen",(typeof t.getTarget==="function"?"bridge":t===window?"window-direct":"element")+":"+e,performance.now()-started);return()=>t.removeEventListener(e,n)}')
replace('e.push(l),t?.addEventListener(s,l[5],{capture:a,passive:l[3]})}',
'e.push(l);if(t){const started=performance.now();t.addEventListener(s,l[5],{capture:a,passive:l[3]});puiBindingCount("nativeBridge",(t===window?"window":"other")+":"+s+(a?":capture":""),performance.now()-started)}}')
replace('n.push(g),f?.addEventListener("abort",g.cleanup,{once:!0}),n.filter',
'n.push(g),puiBindingCount("protoDemand",(t.hostTarget instanceof HTMLElement?"root":"global")+":"+l.inner,0),f?.addEventListener("abort",g.cleanup,{once:!0}),n.filter')
replace('const ge=Of({protoBus:n,hostTarget:o,isEnabled:a,onProtoDemandChange:J});return{rootTarget:te,',
'const ge=Of({protoBus:n,hostTarget:o,isEnabled:a,onProtoDemandChange:J});puiRouterBindingCheckpoint();return{rootTarget:te,')
after=prefix+source
asset.write_text(after)
(q/'asset-binding.json').write_text(json.dumps(dict(asset=asset.name,original=sha256(before.encode()).hexdigest(),observed=sha256(after.encode()).hexdigest(),scope='Emitted router listen and bridge native registration census with separate admitted subscription counts. Nested times may overlap. No binding selection or delivery changes.'),indent=2)+'\n')
