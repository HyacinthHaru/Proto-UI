def once(source,old,new):
    assert source.count(old)==1,old
    return source.replace(old,new)
prefix=r"""
const __s4PassiveObserver=(()=>{
  let active=null,count=0;
  const sequence=['ordinary','beforeActivation','afterActivation','afterActivation','beforeActivation','ordinary'];
  const ordinals=new Map();
  const totals={}, firstSlow=[];
  const windowNow=()=>performance.now();
  const measure=(phase,operation)=>{
    const row=active;
    if(!row) return operation();
    const begin=windowNow();
    try {return operation();}
    finally {(row.phases[phase] ??= []).push(windowNow()-begin);}
  };
  const select=(generation,runtime)=>{
    const ordinal=ordinals.get(runtime) ?? 0;
    if(generation===1) ordinals.set(runtime,ordinal+1);
    const arm=generation===1 ? sequence[ordinal % sequence.length] : 'later';
    return {generation,runtime,arm,begin:windowNow(),phases:{},earlyAnchor:null,hadEarlyRead:false};
  };
  const publication=(row,operation)=>{
    const {generation,runtime,arm}=row;
    const previous=active;
    active=row;
    try {return operation();}
    finally {
      active=previous;
      const end=windowNow();
      const group=(totals[runtime] ??= {})[arm] ??= {calls:0,elapsed:0,phases:{},nonemptyEarly:0,nonemptyOriginal:0,anchorChanged:0};
      if(row.earlyAnchor) group.nonemptyEarly++;
      if(row.originalAnchor) group.nonemptyOriginal++;
      if(row.hadEarlyRead && row.earlyAnchor!==row.originalAnchor) group.anchorChanged++;
      delete row.earlyAnchor;
      delete row.originalAnchor;
      group.calls++;group.elapsed+=end-row.begin;
      for(const [phase,times] of Object.entries(row.phases)){
        const data=group.phases[phase] ??= {calls:0,ms:0,max:0};
        for(const time of times){data.calls++;data.ms+=time;data.max=Math.max(time,data.max);}
      }
      if(end-row.begin>=50 && firstSlow.length<10) firstSlow.push({...row,end});
      count++;
      if((count & (count-1))===0 || count%64===0){
        console.info('[S4 activation preflight]',JSON.stringify({count,pageMs:end,totals,firstSlow}));
      }
    }
  };
  const probe=(phase,doc)=>{
    if(active?.arm!==phase) return;
    active.earlyAnchor=measure('extraAnchor',()=>doc.getSelection()?.anchorNode);
    active.hadEarlyRead=true;
  };
  const original=(operation)=>{
    const value=measure('anchorNode',operation);
    if(active) active.originalAnchor=value;
    return value;
  };
  const activate=(row,doc,operation)=>{
    if(!row) return operation();
    const previous=active;
    active=row;
    try {
      probe('beforeActivation',doc);
      const value=measure('activate',operation);
      probe('afterActivation',doc);
      return value;
    } finally {active=previous;}
  };
  return {measure,publication,select,activate,original};
})();
globalThis.__puiS4PassiveObserver=__s4PassiveObserver;
"""
