  const windowNow=()=>performance.now();
  const measure=(phase,operation)=>{
    const row=active;
    if(!row) return operation();
    const begin=windowNow();
    try {return operation();}
    finally {(row.phases[phase] ??= []).push(windowNow()-begin);}
  };
  const publication=(generation,operation)=>{
    const previous=active;
    const row={generation,begin:windowNow(),phases:{}};
    active=row;
    try {return operation();}

      const end=windowNow();
      const group=totals[generation===1 ? 'generation1' : 'later'] ??= {calls:0,elapsed:0,phases:{}};

from html.parser import HTMLParser
asset=dist/'liquid-library-card.DyhCTi_m.css'
raw=asset.read_bytes()
text=raw.decode()
old='.site-page-frame:has([data-docs-site-header]) main'
new='.site-page-frame:has(>header>[data-docs-site-header]) main'
assert text.count(old)==2
asset.write_text(text.replace(old,new))
class HeaderShape(HTMLParser):
    void={'img','input','meta','link','br','hr','source','base','col','wbr','embed','area','param','track'}
    def __init__(self):super().__init__();self.open=[];self.matches=[]
    def handle_starttag(self,name,pairs):
        attrs=dict(pairs)
        if 'data-docs-site-header' in attrs:
            self.matches.append(len(self.open)>1 and self.open[-1][0]=='header' and 'site-page-frame' in self.open[-2][1].get('class','').split())
        if name not in self.void:self.open.append((name,attrs))
    def handle_endtag(self,name):
        for i in range(len(self.open)-1,-1,-1):
            if self.open[i][0]==name:del self.open[i:];break
pages=dist.parent
count=0
for html in pages.rglob('index.html'):
    parser=HeaderShape()
    parser.feed(html.read_text())
    assert all(parser.matches),html
    count+=len(parser.matches)
assert count>600,count
(out/'css-binding.json').write_text(json.dumps({'file':asset.name,'original':sha256(raw).hexdigest(),'control':sha256(asset.read_bytes()).hexdigest(),'exactChanges':2,'old':old,'new':new,'compiledDocHeadersChecked':count,'scope':'Only two built stylesheet selectors changed.
