    before=p.read_text()
    after=transform(before)
    assert after!=before
    p.write_text(after)
    assets[p.name]=dict(original=sha256(before.encode()).hexdigest(),observed=sha256(after.encode()).hexdigest())

    assert source.count(old)==1,(old,source.count(old))
    return source.replace(old,new,1)
