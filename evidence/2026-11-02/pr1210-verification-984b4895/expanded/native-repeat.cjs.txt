const {createRequire}=require('node:module');const fs=require('node:fs');const path=require('node:path');const assert=require('node:assert/strict');const {chromium}=createRequire(path.join(process.cwd(),'apps/www/package.json'))('playwright-core');const out=process.argv[2] || '$EVIDENCE_ROOT/native';fs.mkdirSync(out,{recursive:true});
(async()=>{const browser=await chromium.launch({headless:true,executablePath:process.env.CHROME_PATH});const results=[];try{for(const family of ['shadcn','brutalist']){const row={family,steps:[],errors:[]};results.push(row);const context=await browser.newContext({viewport:{width:1440,height:960},deviceScaleFactor:1});const page=await context.newPage();page.on('pageerror',e=>row.errors.push(String(e)));try{
const route=family==='shadcn'?'/zh-cn/ui-libraries/shadcn/button/':'/zh-cn/ui-libraries/brutalist/components/button/';await page.goto('http://127.0.0.1:51480'+route);const root=page.locator(`[data-site-header] [data-adapter-select] wc-${family}-select-root`);const trigger=root.getByRole('combobox');await trigger.waitFor({state:'visible'});await trigger.click();await page.getByRole('option').first().waitFor({state:'visible'});await page.keyboard.press('ArrowDown');
async function read(){return page.evaluate(family=>{const root=document.querySelector(`[data-site-header] [data-adapter-select] wc-${family}-select-root`);const trigger=root.querySelector('[role="combobox"]');const popup=document.getElementById(trigger.getAttribute('aria-controls'));const items=[...popup.querySelectorAll(`wc-${family}-select-item`)];let focused=document.activeElement;while(focused?.shadowRoot?.activeElement)focused=focused.shadowRoot.activeElement;return{open:root.getExposes().open.get(),value:root.getExposes().value.get(),focusedValue:focused?.closest(`[data-value]`)?.getAttribute('data-value'),active:items.map(e=>({value:e.getAttribute('data-value'),active:e.getExposes().active.get(),selected:e.getExposes().selected.get()})),focusHTML:focused?.outerHTML.slice(0,300)}},family)}
const before=await read();row.steps.push({stage:'before',facts:before,aria:await page.getByRole('listbox').ariaSnapshot()});assert.equal(before.open,true);assert.equal(before.active.filter(x=>x.active).length,1);assert.ok(before.focusedValue);await page.screenshot({path:path.join(out,family+'-before.png')});
row.returnValue=await root.evaluate(el=>{window.__repeatFocus=document.activeElement;window.__repeatRoot=el;window.__repeatEvents=[];el.addEventListener('openChange',e=>window.__repeatEvents.push(e.detail));return el.getExposes().requestOpen({open:true,reason:'native-consumer-repeat-validation',focusReason:'keyboard',entry:'selected-or-first'})});await page.evaluate(()=>new Promise(r=>requestAnimationFrame(()=>requestAnimationFrame(r))));const after=await read();row.steps.push({stage:'after',facts:after,aria:await page.getByRole('listbox').ariaSnapshot()});row.events=await page.evaluate(()=>window.__repeatEvents);assert.deepEqual(after,before);assert.equal(await page.evaluate(()=>document.activeElement===window.__repeatFocus),true);await page.screenshot({path:path.join(out,family+'-after.png')});await page.keyboard.press('Escape');assert.equal(await trigger.getAttribute('aria-expanded'),'false');row.status='passed';
}catch(e){row.status='failed';row.failure=String(e);await page.screenshot({path:path.join(out,family+'-failure.png')}).catch(()=>{});}finally{await context.close();fs.writeFileSync(path.join(out,family+'.json'),JSON.stringify(row,null,2));}}}finally{await browser.close();fs.writeFileSync(path.join(out,'summary.json'),JSON.stringify(results,null,2));}console.log(results);process.exitCode=results.some(x=>x.status!=='passed')?1:0;})().catch(e=>{console.error(e);process.exitCode=1});
